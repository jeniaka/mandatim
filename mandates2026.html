// proxy.js — node proxy.js  (then set prefix to http://localhost:8787/fetch?url=)
const http = require('http'), { URL } = require('url');

http.createServer(async (req, res) => {
  res.setHeader('Access-Control-Allow-Origin', '*');
  const target = new URL(req.url, 'http://x').searchParams.get('url');
  if (!target) { res.writeHead(400).end('missing url'); return; }
  try {
    const r = await fetch(target, { headers: { 'User-Agent': 'mandates-dashboard/1.0' } });
    res.writeHead(r.status, { 'Content-Type': r.headers.get('content-type') || 'application/json' });
    res.end(await r.text());
  } catch (e) { res.writeHead(502).end(String(e)); }
}).listen(8787, () => console.log('proxy on :8787'));
